\documentclass[tikz,border=12pt]{standalone}

\definecolor{bluefill}{HTML}{E7F5FF}
\definecolor{blueedge}{HTML}{1971C2}
\definecolor{pinkfill}{HTML}{FFF0F6}
\definecolor{pinkedge}{HTML}{C2255C}
\definecolor{greenfill}{HTML}{EBFBEE}
\definecolor{greenedge}{HTML}{2B8A3E}
\definecolor{textgray}{HTML}{343A40}

\tikzset{
  box/.style={
    draw,
    rounded corners=3pt,
    minimum width=3.4cm,
    minimum height=1.2cm,
    align=center,
    font=\small,
    text=textgray
  },
  arrow/.style={-{Latex[length=2.8mm]}, thick, textgray}
}

\begin{document}
\begin{tikzpicture}[node distance=1.25cm and 1.45cm]
  \node[box, fill=bluefill, draw=blueedge] (private) {Private scalar\\$d$};
  \node[box, fill=bluefill, draw=blueedge, right=of private] (public)
    {Public point\\$Q=dG$};

  \node[box, fill=pinkfill, draw=pinkedge, below=of private] (sign)
    {Sign with $d$, digest $z$,\\and fresh nonce $k$};
  \node[box, fill=pinkfill, draw=pinkedge, right=of sign] (signature)
    {Signature\\$(r,s)$};

  \node[box, fill=greenfill, draw=greenedge, below=of signature] (verify)
    {Compute\\$X=zs^{-1}G+rs^{-1}Q$};
  \node[box, fill=greenfill, draw=greenedge, left=of verify] (result)
    {Accept when\\$x(X)\bmod n=r$};

  \draw[arrow] (private) -- node[above, font=\scriptsize] {scalar multiplication} (public);
  \draw[arrow] (private) -- (sign);
  \draw[arrow] (sign) -- (signature);
  \draw[arrow] (signature) -- (verify);
  \draw[arrow] (public.east) -- ++(0.55,0) |- (verify.east);
  \draw[arrow] (verify) -- (result);

  \node[draw=greenedge, dashed, rounded corners=4pt, fit=(verify)(result),
    inner sep=8pt, label={[font=\small, text=greenedge]below:Verification reconstructs $X=kG$}] {};
\end{tikzpicture}
